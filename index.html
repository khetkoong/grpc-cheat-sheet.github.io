🏗Struct & Methodใช้ทุกวัน
Struct คือการนิยาม data type เหมือน class
Value receiver (u User) — อ่านอย่างเดียว ไม่แก้ struct ต้นฉบับ
Pointer receiver (u *User) — แก้ไข struct ได้ ใช้บ่อยกว่ามาก
type User struct {
  ID    string
  Email string
  Age   int
}

// value receiver — copy ค่ามา อ่านได้อย่างเดียว
func (u User) Greet() string {
  return "Hello, " + u.Email
}

// pointer receiver — แก้ไข struct ต้นฉบับได้
func (u *User) SetEmail(e string) {
  u.Email = e
}
🔌Interfaceใช้ทุกวัน
หัวใจของ Go — ไม่ต้อง declare ว่า implement แค่มี method ครบก็เป็น interface นั้นทันที (duck typing)
ใช้คู่กับ Hexagonal เพื่อนิยาม Port ทุกตัว
type UserRepo interface {
  FindByID(id string) (*User, error)
  Save(u *User) error
}

// implement โดยไม่ต้องเขียน "implements UserRepo"
type postgresRepo struct{ db *sql.DB }

func (r *postgresRepo) FindByID(id string) (*User, error) {
  // query postgres...
}
func (r *postgresRepo) Save(u *User) error {
  // insert...
}
📦Slice & Mapใช้ทุกวัน
slice = dynamic array, map = hash map
comma-ok pattern val, ok := m[key] — เช็คก่อนใช้ ไม่งั้น map จะ return zero value แทน panic
range ใช้ loop ทั้ง slice และ map
// slice
ids := []string{"a", "b"}
ids = append(ids, "c")
for i, v := range ids { /* i=index, v=value */ }

// map
cache := map[string]*User{}
cache["u1"] = &User{ID: "u1"}

// comma-ok: ตรวจว่า key มีอยู่จริงไหม
val, ok := cache["u1"]
if !ok {
  // ไม่พบ key นี้
}
_ = val
🏭Constructor Patternใช้ทุกวัน
Go ไม่มี constructor — ใช้ function New...() แทน
รับ dependencies เข้ามาเป็น interface (manual DI) — เป็น pattern หลักของ Hexagonal
เก็บ dependencies เป็น unexported field
type Service struct {
  repo   UserRepo     // interface ไม่ใช่ concrete type
  logger *slog.Logger
}

func NewService(
  repo   UserRepo,
  logger *slog.Logger,
) *Service {
  return &Service{repo: repo, logger: logger}
}
📋Defer & Cleanupใช้ทุกวัน
defer รันโค้ดตอน function return เสมอ ไม่ว่าจะ return ปกติหรือ error
ป้องกัน resource leak — ปิด file, unlock mutex, rollback transaction
stack-based: defer ตัวหลังสุดรันก่อน
func doWork(path string) error {
  f, err := os.Open(path)
  if err != nil { return err }
  defer f.Close() // รันตอน return เสมอ

  mu.Lock()
  defer mu.Unlock() // ไม่มีทางลืม unlock

  tx, _ := db.Begin()
  defer tx.Rollback() // safe fallback ก่อน commit
  // ... do work ...
  return tx.Commit()
}
🧩Type Embeddingใช้บ้าง
Embed struct เพื่อ "inherit" fields และ methods โดยไม่ต้องเขียนซ้ำ
ใช้บ่อยตอนทำ base entity (ID, CreatedAt) หรือ embed UnimplementedXxxServer ของ gRPC
type Base struct {
  ID        string
  CreatedAt time.Time
}

type Post struct {
  Base            // embed — ได้ ID, CreatedAt ฟรี
  Title   string
  Content string
}

p := Post{}
p.ID = "p1"      // access Base field ได้เลย
⚙️Functional Optionsใช้บ้าง
Pattern สำหรับ config ที่มี optional parameter หลายตัว
เจอบ่อยใน library ดีๆ เช่น grpc.NewServer(opt1, opt2) — ตัวเองเขียนไม่บ่อยนักแต่ต้องอ่านออก
type Option func(*Config)

func WithTimeout(d time.Duration) Option {
  return func(c *Config) { c.Timeout = d }
}

func New(opts ...Option) *Client {
  cfg := &Config{Timeout: 30}
  for _, o := range opts { o(cfg) }
  return &Client{cfg: cfg}
}
Concurrencygoroutine · channel · sync
🚀Goroutine & Channelใช้ทุกวัน
goroutine = lightweight thread ของ Go, go func() รัน async ทันที
channel = ท่อส่งข้อมูลระหว่าง goroutine อย่างปลอดภัย
buffered = ไม่ block จนกว่าจะเต็ม, unbuffered = block จนมีคนรับ
ch := make(chan int, 1) // buffered จุ 1

go func() {
  ch <- compute() // ส่งค่าเข้า channel
}()

result := <-ch // block รอจนได้ค่า

// fan-out: ส่งงานหลายชิ้นพร้อมกัน
func fanOut(jobs []Job) []Result {
  ch := make(chan Result, len(jobs))
  for _, j := range jobs {
    go func(j Job) { ch <- process(j) }(j)
  }
  res := make([]Result, len(jobs))
  for i := range res { res[i] = <-ch }
  return res
}
⏱Context & WaitGroupใช้ทุกวัน
context ส่ง timeout/cancel ผ่าน call chain — ต้องส่งต่อทุก function เสมอ
WaitGroup รอให้ทุก goroutine ทำเสร็จก่อน
select = switch สำหรับ channel รับหลาย channel พร้อมกัน
ctx, cancel := context.WithTimeout(
  context.Background(), 5*time.Second,
)
defer cancel() // ต้อง defer cancel เสมอ

var wg sync.WaitGroup
for _, item := range items {
  wg.Add(1)
  go func(item Item) {
    defer wg.Done()
    process(ctx, item)
  }(item)
}
wg.Wait()

select {
case v := <-dataCh: handle(v)
case <-ctx.Done():  return ctx.Err()
}
🔐Mutex & RWMutexใช้บ้าง
ใช้เมื่อหลาย goroutine access ข้อมูลเดียวกันพร้อมกัน
RWMutex ดีกว่าถ้า read เยอะกว่า write — อ่านพร้อมกันได้หลาย goroutine แต่ write ต้องรอ
ถ้า design ด้วย channel ได้ มักไม่ต้องใช้ mutex เลย
type SafeCache struct {
  mu    sync.RWMutex
  store map[string]any
}

// RLock: read พร้อมกันได้หลาย goroutine
func (c *SafeCache) Get(k string) any {
  c.mu.RLock(); defer c.mu.RUnlock()
  return c.store[k]
}

// Lock: write ได้ครั้งละตัวเดียว
func (c *SafeCache) Set(k string, v any) {
  c.mu.Lock(); defer c.mu.Unlock()
  c.store[k] = v
}
🏃errgroup — Parallel Tasksใช้บ้าง
รัน goroutines parallel แล้วรอทุกตัว — ถ้ามี error ตัวไหนจะ cancel ทุกตัวให้อัตโนมัติ
เหมาะมากสำหรับ parallel fetch เช่น ดึงข้อมูลจาก 2-3 service พร้อมกัน แทนที่จะรอทีละตัว
import "golang.org/x/sync/errgroup"

func fetchAll(ctx context.Context) error {
  g, ctx := errgroup.WithContext(ctx)

  g.Go(func() error { return fetchUsers(ctx) })
  g.Go(func() error { return fetchPosts(ctx) })

  // รอทุกตัว — return error แรกที่เจอ
  return g.Wait()
}
Error Handlingidiomatic go errors
🚨Custom Error & Sentinelใช้บ้าง
สร้าง error type ที่มี context เพิ่ม เช่น resource ชื่ออะไร ID อะไร
Sentinel error คือ error ที่ประกาศไว้แค่ครั้งเดียว ใช้เปรียบเทียบด้วย errors.Is()
ตั้งชื่อขึ้นต้น Err เสมอ เช่น ErrNotFound
type NotFoundError struct {
  Resource string
  ID       string
}
func (e *NotFoundError) Error() string {
  return fmt.Sprintf("%s %s not found", e.Resource, e.ID)
}

// sentinel — ประกาศที่ package level
var (
  ErrUnauthorized = errors.New("unauthorized")
  ErrInvalidInput = errors.New("invalid input")
)
🔍errors.Is / As / %wใช้ทุกวัน
errors.Is — เช็ค sentinel error (unwrap chain ทั้งหมด)
errors.As — ดึง type ออกจาก error chain
%w — wrap error พร้อมเพิ่ม context เพื่อ trace กลับไปได้ เป็น pattern หลักที่ใช้ทุก function
err := getUser(ctx, id)

// Is: เช็ค sentinel (ทะลุ wrap chain)
if errors.Is(err, ErrUnauthorized) { /* 401 */ }

// As: ดึง custom error type ออกมา
var nf *NotFoundError
if errors.As(err, &nf) {
  fmt.Println("missing:", nf.ID)
}

// %w: wrap + เพิ่ม context ให้ trace ได้
return fmt.Errorf("getUser: %w", err)

🛡Interceptor (Middleware)ใช้ทุกวัน
เทียบกับ HTTP middleware — รันก่อน/หลัง handler ทุกตัว
ใช้สำหรับ auth, logging, panic recovery, tracing
Chain interceptors ได้ด้วย ChainUnaryInterceptor — เรียงลำดับสำคัญ
func loggingInterceptor(
  ctx context.Context, req any,
  info *grpc.UnaryServerInfo,
  handler grpc.UnaryHandler,
) (any, error) {
  start := time.Now()

  resp, err := handler(ctx, req) // เรียก handler จริง

  slog.Info("gRPC",
    "method", info.FullMethod,
    "dur", time.Since(start),
    "err", err,
  )
  return resp, err
}
⚠️gRPC Status Errorsใช้ทุกวัน
ห้าม return Go error ธรรมดาจาก handler — ต้อง wrap ด้วย status.Errorf() เสมอ
แปลง domain error → gRPC code ที่ handler layer เป็น responsibility ของ Primary Adapter
func (s *server) GetUser(
  ctx context.Context, req *pb.GetUserRequest,
) (*pb.UserResponse, error) {

  user, err := s.repo.FindByID(ctx, req.Id)
  if errors.Is(err, ErrNotFound) {
    return nil, status.Errorf(
      codes.NotFound, "user %s not found", req.Id,
    )
  }
  if err != nil {
    return nil, status.Error(codes.Internal, err.Error())
  }
  return toProto(user), nil
}

🏷Metadata (Headers)ใช้บ้าง
Metadata คือ HTTP headers ของ gRPC — ใช้ส่ง auth token, request-id, trace id
ทำใน interceptor แทน จะได้ไม่ต้องเขียนทุก call
Outgoing = ส่งออกจาก client, Incoming = รับที่ server
import "google.golang.org/grpc/metadata"

// client: เพิ่ม metadata ก่อน call
md := metadata.Pairs(
  "authorization", "Bearer "+token,
  "x-request-id",  requestID,
)
ctx = metadata.NewOutgoingContext(ctx, md)

// server: รับ metadata จาก request
md, ok := metadata.FromIncomingContext(ctx)
if ok {
  tokens := md.Get("authorization")
  _ = tokens
}
Hexagonal Architectureports & adapters — concept
🎯Domain Coreใช้ทุกวัน
ตรงกลางสุด — รู้จักแค่ business rules ไม่รู้จักโลกภายนอกเลย

🔌Ports (Interfaces)ใช้ทุกวัน
Port คือ interface ที่ Domain กำหนดเอง — "ฉันต้องการอะไร" โดยไม่สนว่าใครจะมา implement
Input port = use case interface (Primary เรียก), Output port = dependency (Secondary implement)
Input Port — use case interface
ถูก implement โดย Use Case

Output Port — dependency interface
ถูก implement โดย Secondary Adapter
🔄Adaptersใช้ทุกวัน
แปลงภาษาระหว่าง domain กับโลกภายนอก — swap ได้โดยไม่แตะ domain
Primary = รับ request เข้า, Secondary = ถูก domain เรียกออก
Primary (Driving)
gRPC handler HTTP handler CLI

Secondary (Driven)
postgres repo redis
FCM adapter kafka producer

🔌Port — Output Interfaceใช้ทุกวัน
Domain กำหนด interface นี้เพื่อบอกว่า "ฉันต้องการอะไร" — implementation จริงอยู่ที่ adapter
ทำให้ test use case ได้โดย inject mock แทน real adapter ไม่ต้องต่อ DB หรือ Firebase จริง
// internal/port/output.go
package port

type NotiRepository interface {
  Save(ctx context.Context, n *domain.Notification) error
  FindByUser(ctx context.Context, userID string) (
    []*domain.Notification, error)
}

type PushPayload struct {
  Title string
  Body  string
  Data  map[string]string
}

type PushNotifier interface {
  Send(ctx context.Context, userID string, p PushPayload) error
  Broadcast(ctx context.Context, topic string, p PushPayload) error
}
⚙️Use Caseใช้ทุกวัน
Business logic อยู่ที่นี่ทั้งหมด — ไม่รู้จัก gRPC, Postgres, Firebase
รู้จักแค่ port (interface) — test ได้ 100% โดยไม่ต้อง infra จริง
// internal/usecase/broadcast.go
type BroadcastUseCase struct {
  repo port.NotiRepository
  push port.PushNotifier
}

func (uc *BroadcastUseCase) Execute(
  ctx context.Context, req BroadcastInput,
) error {
  n := &domain.Notification{
    ID: uuid.NewString(), RefID: req.PostID,
  }
  // ไม่รู้ว่า repo เป็น Postgres หรืออะไร
  if err := uc.repo.Save(ctx, n); err != nil {
    return fmt.Errorf("save: %w", err)
  }
  // ไม่รู้ว่า push เป็น FCM หรืออะไร
  return uc.push.Broadcast(ctx, "all-users", port.PushPayload{
    Title: req.Title, Body: req.Body,
    Data: map[string]string{"postId": req.PostID},
  })
}

อยากเปลี่ยน provider แค่สร้าง adapter ใหม่ implement port.PushNotifier — ไม่แตะ use case เลย
// internal/adapter/fcm/adapter.go
type FCMAdapter struct{ client *messaging.Client }

// implements port.PushNotifier
func (a *FCMAdapter) Broadcast(
  ctx context.Context,
  topic string, p port.PushPayload,
) error {
  // แปลง domain payload → Firebase message
  _, err := a.client.Send(ctx, &messaging.Message{
    Topic: topic,
    Notification: &messaging.Notification{
      Title: p.Title, Body: p.Body,
    },
    Data: p.Data,
  })
  return err
}

🔧Dependency Injection — main.goใช้ทุกวัน
สร้าง adapter แล้วส่งเข้า use case ทีเดียว — use case รับแค่ interface ไม่รู้ว่าเป็น concrete type อะไร
ทำครั้งเดียวตอน startup — ถ้า dependency เปลี่ยน แก้แค่ที่นี่ที่เดียว
func main() {
  // 1. infra — concrete implementations
  db  := connectPostgres(cfg.DBURL)
  fcm := initFirebase(cfg.Credentials)

  // 2. secondary adapters
  notiRepo := postgres.NewNotiRepo(db)
  pushSvc  := fcmadapter.New(fcm)

  // 3. use cases (รับแค่ interface)
  broadcastUC := usecase.NewBroadcastUseCase(notiRepo, pushSvc)

  // 4. primary adapter
  grpcHandler := grpcadapter.NewHandler(broadcastUC)

  // 5. start server
  s := grpc.NewServer(...)
  pb.RegisterNotificationServiceServer(s, grpcHandler)
  s.Serve(lis)
}
🎯gRPC Handler (Primary Adapter)ใช้ทุกวัน
แปลง proto request → domain input → เรียก use case → แปลง result กลับเป็น proto
handler ไม่มี business logic เลย — แค่ translate และ delegate ไป use case
// internal/adapter/grpc/handler.go
type Handler struct {
  pb.UnimplementedNotificationServiceServer
  broadcastUC port.BroadcastUseCase
}

func (h *Handler) BroadcastNotification(
  ctx context.Context, req *pb.BroadcastRequest,
) (*pb.BroadcastResponse, error) {

  // proto → domain input (translate ที่นี่เท่านั้น)
  err := h.broadcastUC.Execute(ctx, usecase.BroadcastInput{
    Topic: req.Topic, Title: req.Title,
    Body: req.Body,   PostID: req.PostId,
  })
  if err != nil {
    return nil, status.Error(codes.Internal, err.Error())
  }
  return &pb.BroadcastResponse{Status: okStatus()}, nil
}
🧪Unit Test — Mock Portใช้ทุกวัน
ข้อดีสูงสุดของ Hexagonal — test use case ได้โดยไม่ต้องต่อ DB หรือ Firebase จริงเลย
สร้าง mock ที่ implement port แล้ว inject เข้าไป — test รันเร็วมาก deterministic 100%
// mock implement port.PushNotifier
type mockPush struct{ calls []string }
func (m *mockPush) Broadcast(
  ctx context.Context, topic string, p port.PushPayload,
) error {
  m.calls = append(m.calls, topic)
  return nil
}

func TestBroadcast_Success(t *testing.T) {
  push := &mockPush{}
  uc   := usecase.NewBroadcastUseCase(&mockRepo{}, push)

  err := uc.Execute(ctx, BroadcastInput{
    Topic: "all-users", Title: "New Post",
  })

  assert.NoError(t, err)
  assert.Equal(t, "all-users", push.calls[0])
}

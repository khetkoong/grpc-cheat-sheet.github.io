🚀 Goroutine & Channel 
func main() {
  ch := make(chan int, 1) // buffered

  go func() {
    ch <- compute()
  }()

  result := <-ch // block until ready
  fmt.Println(result)
}

// fan-out pattern
func fanOut(jobs []Job) []Result {
  ch := make(chan Result, len(jobs))
  for _, j := range jobs {
    go func(j Job) { ch <- process(j) }(j)
  }
  res := make([]Result, len(jobs))
  for i := range res { res[i] = <-ch }
  return res
}

🔐 Mutex & sync.Map 
type SafeCache struct {
  mu    sync.RWMutex
  store map[string]any
}

func (c *SafeCache) Get(k string) any {
  c.mu.RLock()
  defer c.mu.RUnlock()
  return c.store[k]
}

func (c *SafeCache) Set(k string, v any) {
  c.mu.Lock()
  defer c.mu.Unlock()
  c.store[k] = v
}

🛡 Interceptor (Middleware) 
func loggingInterceptor(
  ctx context.Context,
  req any,
  info *grpc.UnaryServerInfo,
  handler grpc.UnaryHandler,
) (any, error) {
  start := time.Now()

  resp, err := handler(ctx, req)

  slog.Info("gRPC",
    "method", info.FullMethod,
    "duration", time.Since(start),
    "error", err,
  )
  return resp, err
}
⚠️ gRPC Status Errors 
import (
  "google.golang.org/grpc/codes"
  "google.golang.org/grpc/status"
)

func (s *server) GetUser(
  ctx context.Context,
  req *pb.GetUserRequest,
) (*pb.UserResponse, error) {

  user, err := s.repo.FindByID(ctx, req.Id)
  if errors.Is(err, ErrNotFound) {
    return nil, status.Errorf(
      codes.NotFound, "user %s not found", req.Id,
    )
  }
  if err != nil {
    return nil, status.Error(codes.Internal, err.Error())
  }
  return toProto(user), nil
}

⚙️ Use Case 
// internal/usecase/broadcast.go
type BroadcastUseCase struct {
  repo port.NotiRepository
  push port.PushNotifier
}

func (uc *BroadcastUseCase) Execute(
  ctx context.Context,
  req BroadcastInput,
) error {
  // 1. build domain object
  n := &domain.Notification{...}
  
  // 2. save to repo
  if err := uc.repo.Save(ctx, n); err != nil {
    return fmt.Errorf("save noti: %w", err)
  }
  
  // 3. push via FCM (through port)
  return uc.push.Broadcast(ctx, "all-users", PushPayload{
    Title: req.Title,
    Body:  req.Body,
    Data:  map[string]string{"postId": req.PostID},
  })
}

  client *messaging.Client
}

// implements port.PushNotifier
func (a *FCMAdapter) Broadcast(
  ctx context.Context,
  topic string,
  p port.PushPayload,
) error {
  _, err := a.client.Send(ctx, &messaging.Message{
    Topic: topic,
    Notification: &messaging.Notification{
      Title: p.Title,
      Body:  p.Body,
    },
    Data: p.Data,
  })
  return err
}

🎯 gRPC Handler (Primary Adapter) 
// internal/adapter/grpc/handler.go
type Handler struct {
  pb.UnimplementedNotificationServiceServer
  broadcastUC port.BroadcastUseCase
}

func (h *Handler) BroadcastNotification(
  ctx context.Context,
  req *pb.BroadcastRequest,
) (*pb.BroadcastResponse, error) {

  // proto → domain input
  err := h.broadcastUC.Execute(ctx, usecase.BroadcastInput{
    Topic:  req.Topic,
    Title:  req.Title,
    Body:   req.Body,
    PostID: req.PostId,
  })
  if err != nil {
    return nil, status.Error(codes.Internal, err.Error())
  }
  return &pb.BroadcastResponse{
    Status: okStatus(),
  }, nil
}
